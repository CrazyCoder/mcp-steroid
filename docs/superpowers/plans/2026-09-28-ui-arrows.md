# UI class split, then highlight arrows, per-step numbers and styles

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Split the largest `steroid_ui` classes into focused files without a
behavior change, then add arrows, `"number": false` and `style` to screenshot
highlights as the spec describes.

**Architecture:** The split comes first so the arrow work lands in small files.
The server's step model, highlight parsing and step validation separate. The
plugin's picture code (`UiCapture`) separates from mark layout and drawing
(`UiCallouts`), and `UiSession` hands its screenshot, highlight, row and window
steps to classes that reach the session through one interface,
`UiStepContext`. Arrows then extend `UiHighlight` on the server and
`UiCallouts.Mark` in the plugin; placement is a pure function tested without
an IDE.

**Tech Stack:** Kotlin, IntelliJ Platform, kotlinx.serialization JSON, JUnit 4,
Gradle; Node.js for JetDesk's `capture.js`.

**Spec:** `docs/superpowers/specs/2026-09-28-ui-arrows-design.md`

## Review Focus

1. A joined click pointer (a right click on a word) with `arrow`: the pointer stays bare and draws no arrow, as it has no callout of its own; the report says it joined. Test in Task 7.
2. A highlight near the picture's edge with `from: auto` and all sides blocked: the arrow still draws inside the picture, on the spot that covers least, never off the canvas. Test in Task 7.
3. `style` on the step and on a highlight: a highlight that sets only `width` keeps the step's `color`. Test in Task 5.
4. `"crop":"highlights"` with an arrow at 2x length: the crop holds the whole callout. Test in Task 6 (`markArea`).
5. A picture where every highlight has `"number": false`: no badges at all and the report says "outlined without numbers". Test in Task 6.

```kotlin
/** The side of its target an arrow's callout sits on, as a direction from the target: [dx] and [dy] are -1, 0 or 1. */
enum class UiArrowSide(val wire: String, val dx: Int, val dy: Int) {
    AUTO("auto", 0, 0), LEFT("left", -1, 0), RIGHT("right", 1, 0), ABOVE("above", 0, -1), BELOW("below", 0, 1),
    ABOVE_LEFT("above-left", -1, -1), ABOVE_RIGHT("above-right", 1, -1), BELOW_LEFT("below-left", -1, 1), BELOW_RIGHT("below-right", 1, 1);

    val opposite: UiArrowSide get() = entries.first { it != AUTO && it.dx == -dx && it.dy == -dy }

    companion object {
        /** The order `auto` tries: the four sides, then the diagonals. */
        val AUTO_ORDER = listOf(RIGHT, LEFT, BELOW, ABOVE, BELOW_RIGHT, ABOVE_RIGHT, BELOW_LEFT, ABOVE_LEFT)
        fun of(wire: String) = entries.firstOrNull { it.wire == wire }
    }
}

enum class UiArrowHead(val wire: String) { FILLED("filled"), OPEN("open"), NONE("none") }

/** An arrow from a callout to its highlight. */
data class UiArrow(val from: UiArrowSide = UiArrowSide.AUTO, val length: Int = DEFAULT_LENGTH, val head: UiArrowHead = UiArrowHead.FILLED) {
    companion object {
        const val DEFAULT_LENGTH = 60
        val LENGTHS = 20..400
    }
}

/** The color, as 0xRRGGBB, and line width of marks; a field left null takes the step's, then the default. */
data class UiStyle(val color: Int? = null, val width: Double? = null) {
    /** This style over [base]: each field this one sets wins. */
    fun over(base: UiStyle?): UiStyle = UiStyle(color ?: base?.color, width ?: base?.width)

    companion object {
        val COLORS = linkedMapOf(
            "red" to 0xE52B50, "orange" to 0xFF9F1C, "yellow" to 0xFFD60A, "green" to 0x2A9D5B,
            "blue" to 0x2F6FEB, "purple" to 0x8E44AD, "black" to 0x202020,
        )
        const val DEFAULT_COLOR = 0xE52B50
        const val DEFAULT_WIDTH = 2.5
        val WIDTHS = 1.0..8.0
    }
}
```

```kotlin
object UiArrows {
    const val HEAD_GAP = 3
    private const val GAP = 4

    fun head(box: Rectangle, side: UiArrowSide): Point = Point(
        when (side.dx) { -1 -> box.x - HEAD_GAP; 1 -> box.x + box.width + HEAD_GAP; else -> box.x + box.width / 2 },
        when (side.dy) { -1 -> box.y - HEAD_GAP; 1 -> box.y + box.height + HEAD_GAP; else -> box.y + box.height / 2 },
    )

    fun tail(head: Point, side: UiArrowSide, length: Int): Point {
        val diagonal = side.dx != 0 && side.dy != 0
        val step = if (diagonal) length / sqrt(2.0) else length.toDouble()
        return Point(head.x + (side.dx * step).roundToInt(), head.y + (side.dy * step).roundToInt())
    }

    fun callout(tail: Point, side: UiArrowSide, badge: Int?, labelWidth: Int?, labelHeight: Int): Pair<Rectangle?, Rectangle?> {
        val b = badge?.let { Rectangle(tail.x - it / 2, tail.y - it / 2, it, it) }
        val w = labelWidth ?: return b to null
        val h = b?.height ?: labelHeight
        val y = b?.y ?: (tail.y - h / 2)
        val label = when {
            b != null && side.dx < 0 -> Rectangle(b.x - GAP - w, y, w, h)
            b != null -> Rectangle(b.x + b.width + GAP, y, w, h)
            side.dx < 0 -> Rectangle(tail.x - w, y, w, h)
            side.dx > 0 -> Rectangle(tail.x, y, w, h)
            else -> Rectangle(tail.x - w / 2, if (side.dy < 0) tail.y - h else tail.y, w, h)
        }
        return b to label
    }

    fun shapes(tail: Point, head: Point, headStyle: UiArrowHead, width: Float): Pair<Shape, Shape?> {
        val angle = atan2((head.y - tail.y).toDouble(), (head.x - tail.x).toDouble())
        val len = 4 * width + 6
        val half = 1.5 * width + 2
        fun at(back: Double, side: Double) = Point2D.Double(
            head.x - back * cos(angle) - side * sin(angle), head.y - back * sin(angle) + side * cos(angle),
        )
        // A filled head covers the shaft's end, so the shaft stops at the head's base and its round cap stays hidden.
        val shaftEnd = if (headStyle == UiArrowHead.FILLED) at(len * 0.8, 0.0) else Point2D.Double(head.x.toDouble(), head.y.toDouble())
        val shaft = Line2D.Double(tail.x.toDouble(), tail.y.toDouble(), shaftEnd.x, shaftEnd.y)
        val tip = Point2D.Double(head.x.toDouble(), head.y.toDouble())
        val left = at(len, half)
        val right = at(len, -half)
        val headShape = when (headStyle) {
            UiArrowHead.NONE -> null
            UiArrowHead.OPEN -> Path2D.Double().apply { moveTo(left.x, left.y); lineTo(tip.x, tip.y); lineTo(right.x, right.y) }
            UiArrowHead.FILLED -> Path2D.Double().apply { moveTo(tip.x, tip.y); lineTo(left.x, left.y); lineTo(right.x, right.y); closePath() }
        }
        return shaft to headShape
    }

    fun textOn(fill: Color): Color {
        fun channel(c: Int) = (c / 255.0).let { if (it <= 0.03928) it / 12.92 else ((it + 0.055) / 1.055).pow(2.4) }
        val l = 0.2126 * channel(fill.red) + 0.7152 * channel(fill.green) + 0.0722 * channel(fill.blue)
        // The contrast ratio of white is 1.05 / (l + 0.05) and of black (l + 0.05) / 0.05.
        return if (1.05 / (l + 0.05) >= (l + 0.05) / 0.05) Color.WHITE else Color.BLACK
    }
}
```
